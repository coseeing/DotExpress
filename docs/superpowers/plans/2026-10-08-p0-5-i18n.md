# Shared i18n Module (P0-5) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `client/i18n.py` (`_`, `N_`, `resource_path`, `load_translation`) and make the five first-party modules that build their own gettext catalog (`dialog.py`, `settings/dialogs.py`, `dictionaries/entries.py`, `about.py`, `gui.py`) import from it. Mark the 48 `_MENU_TRANSLATION_MARKERS` entries with `N_` and teach `scripts/generate-pot.bat` the `N_` keyword. No user-visible change, no msgid change, `client/braille/tables/__tables.py` untouched.

**Architecture:** One stateless standard-library module loads the `dotexpress` catalog once at import (fixed `zh_TW`, `fallback=True`) and exposes the bound `gettext` as `_`. Consumers do `from i18n import _` so each keeps a module-global `_` (existing `dialog_module._` / `gui._` test patterns keep working). Task 1 adds the module and its behavior tests. Tasks 2-6 each migrate one module, each starting from a static (AST) test that fails until that module imports from `i18n`; Task 6 also lands the repo-wide "no duplicate setup" guard. Task 7 switches the markers to `N_` and adds the xgettext keyword in one commit. Task 8 is the spec's acceptance run.

**Tech Stack:** Python 3 standard library (`gettext`, `pathlib`, `ast`, `unittest`, `subprocess`). wxPython is not installed; GUI modules load through the P0-4 `tests/wx_stub.py` scopes, which this plan does not touch.

**Spec:** `docs/superpowers/specs/2026-10-08-p0-5-i18n-design.md` (authoritative, latest commit `119de71`; refinements and the one minor factual correction are listed below).

"""Shared gettext setup and bundled-resource lookup for the DotExpress client.

Consumers import the names they use: ``from i18n import _`` (plus ``N_`` and
``resource_path`` where needed). Each consumer then holds the same bound
``gettext`` method in its own module global ``_``, so tests that replace
translations patch the consumer's global (for example ``dialog._``).
Patching ``i18n._`` does not reach modules that have already imported it.
"""

import gettext
import sys
from collections.abc import Sequence
from pathlib import Path

LOCALE_DOMAIN = "dotexpress"
LOCALE_LANGUAGES = ("zh_TW",)
CLIENT_ROOT = Path(__file__).resolve().parent


def resource_path(relative_path: str) -> Path:
    """Return a bundled resource path: the PyInstaller bundle when frozen, else the client directory."""
    if getattr(sys, "frozen", False) and hasattr(sys, "_MEIPASS"):
        base_path = Path(sys._MEIPASS)
    else:
        base_path = CLIENT_ROOT
    return base_path / relative_path


def load_translation(
    localedir: Path | None = None,
    languages: Sequence[str] = LOCALE_LANGUAGES,
) -> gettext.NullTranslations:
    """Load the DotExpress catalog; a missing catalog gives a pass-through translation."""
    if localedir is None:
        localedir = resource_path("locales")
    return gettext.translation(
        LOCALE_DOMAIN,
        localedir=str(localedir),
        languages=list(languages),
        fallback=True,
    )


_translation = load_translation()
_ = _translation.gettext


def N_(message: str) -> str:
    """Mark a string for extraction without translating it."""
    return message
EOF

`$P05/tools/msgid_scan.py`:

```python
"""Throwaway stand-in for xgettext (P0-5). Not committed.

Usage: python -I msgid_scan.py <client_dir> [--save FILE] [--compare FILE]

Collects the first string argument of every _(...) and N_(...) call in
<client_dir>/**/*.py, skipping any path with a .venv or tests directory
(the same file set as scripts/generate-pot.bat), and compares the set with
the msgids of <client_dir>/locales/dotexpress.pot. Exits 1 on any mismatch.
"""

import ast
import json
import os
import sys
from pathlib import Path

KEYWORDS = {"_", "N_"}
SKIP_DIRS = {".venv", "tests"}


def source_msgids(client_dir: Path) -> set[str]:
    found = set()
    for dirpath, dirnames, filenames in os.walk(client_dir):
        dirnames[:] = [d for d in dirnames if d not in SKIP_DIRS]
        for filename in filenames:
            if not filename.endswith(".py"):
                continue
            path = Path(dirpath) / filename
            tree = ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
            for node in ast.walk(tree):
                if (
                    isinstance(node, ast.Call)
                    and isinstance(node.func, ast.Name)
                    and node.func.id in KEYWORDS
                    and node.args
                    and isinstance(node.args[0], ast.Constant)
                    and isinstance(node.args[0].value, str)
                ):
                    found.add(node.args[0].value)
    return found


def _unquote(fragment: str) -> str:
    return ast.literal_eval(fragment)


def pot_msgids(pot_path: Path) -> set[str]:
    msgids = set()
    current = None
    for line in pot_path.read_text(encoding="utf-8").splitlines():
        if line.startswith("msgid "):
            current = [_unquote(line[len("msgid "):])]
        elif line.startswith('"') and current is not None:
            current.append(_unquote(line))
        else:
            if current is not None:
                msgids.add("".join(current))
            current = None
    if current is not None:
        msgids.add("".join(current))
    msgids.discard("")
    return msgids


def main() -> int:
    client_dir = Path(sys.argv[1]).resolve()
    args = sys.argv[2:]
    source = source_msgids(client_dir)
    pot = pot_msgids(client_dir / "locales" / "dotexpress.pot")
    status = 0
    print(f"source msgids: {len(source)}")
    print(f"pot msgids: {len(pot)}")
    if source == pot:
        print("SOURCE == POT")
    else:
        status = 1
        for msgid in sorted(source - pot):
            print(f"only in source: {msgid!r}")
        for msgid in sorted(pot - source):
            print(f"only in pot: {msgid!r}")
    if "--save" in args:
        target = Path(args[args.index("--save") + 1])
        target.write_text(json.dumps(sorted(source), ensure_ascii=False, indent=0), encoding="utf-8")
        print(f"saved {target}")
    if "--compare" in args:
        saved = set(json.loads(Path(args[args.index("--compare") + 1]).read_text(encoding="utf-8")))
        if saved == source:
            print("SOURCE == SAVED")
        else:
            status = 1
            print(f"differs from saved: +{sorted(source - saved)} -{sorted(saved - source)}")
    return status


if __name__ == "__main__":
    sys.exit(main())
```
